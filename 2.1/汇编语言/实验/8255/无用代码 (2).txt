    INT 21H

    MOV AL,LB                  ;将B口起始数据左移再写入B口

DALLY PROC NEAR               ;软件延时子程序 
    PUSH CX 
    PUSH AX 
    MOV CX,0FFFH 
D1: MOV AX,0FFFFH

    JNZ D2 
    LOOP D1 
    POP AX 
    POP CX 
    RET
