#include <bits/stdc++.h>
using namespace std;
int main()
{
    int n, m, x, a[100005], b[100005], s[100005] = {0}, res = 0;
    cin >> n >> m >> x;
    for (int i = 0; i < n; i++)
        cin >> a[i];
    for (int i = 0; i < m; i++)
        cin >> b[i];

    // 双指针，若和大于x，b指针左移，否则a指针右移
    for (int i = 0, j = m - 1; i < n; i++)
    {
        while (j >= 0 && a[i] + b[j] > x)
            j--;
        if (a[i] + b[j] == x){
            cout << i << " " << j << endl;
            // 由于题目给定唯一解，故直接停止
            break;
        }
    }
    return 0;
}
