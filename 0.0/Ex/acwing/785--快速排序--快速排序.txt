#include <iostream>
using namespace std;
// 快速排序--分治
// 确定分界点
// 调整区间
// 递归处理
void quickSort(int *a, int l, int r)
{
    if (l >= r)
        return;
    // 不能每次都选择第一个作为基准数，如果整个数组基本有序那么复杂度相当于冒泡排序O(n^2)
    int pivotIdx = l + rand() % (r - l + 1);
    swap(a[l], a[pivotIdx]);
    int tem = a[l], i = l - 1, j = r + 1;
    //  分出左右
    while (i < j)
    {
        do i++; while (a[i] < tem);
        do j--; while (a[j] > tem);
        if (i < j)
            swap(a[i], a[j]);
    }
    // 递归左右，用什么递归，什么就不能作为边界，即tem不能等于相应的基准
    quickSort(a, l, j);
    quickSort(a, j + 1, r);
}
int main()
{
    int n, a[1000005];
    cin >> n;
    for (int i = 0; i < n; i++)
    {
        cin >> a[i];
    }
    quickSort(a, 0, n - 1);
    for (int i = 0; i < n; i++)
    {
        cout << a[i] << " ";
    }

    return 0;
}
