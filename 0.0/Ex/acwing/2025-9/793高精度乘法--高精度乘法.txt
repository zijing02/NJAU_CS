#include <bits/stdc++.h>
using namespace std;
// 高精度乘法
// 大整数每一位分别乘整一个小乘数
vector<int> mul(vector<int> A, int b)
{
    vector<int> C;
    int t = 0;
    
    // 乘法和加法一样，停止条件要加一个仅为不为0
    for (int i = 0; i < A.size() || t; i++)
    {
        if (i < A.size())
            t += A[i] * b;
        C.push_back(t % 10);
        t /= 10;
    }
    return C;
}
int main()
{
    string a;
    int b;
    cin >> a >> b;
    vector<int> A, C;
    for (int i = a.size() - 1; i >= 0; i--)
        A.push_back(a[i] - '0');
    C = mul(A, b);

    // A*b=0则C的最后一位为0，整个结果为0，直接输出0
    if (C.back() == 0)
        cout << 0;
    else
        for (int i = C.size() - 1; i >= 0; i--)
            cout << C[i];
    return 0;
}
