#include <bits/stdc++.h>
using namespace std;
// 前缀和
// 用空间换时间，预处理出前缀和数组sum，sum[i]表示前i个数的和
// 则区间[l,r]的和为sum[r]-sum[l-1]
// 数组从1开始，目的是为了统一公式
int n, m, l, r, a[100005], sum[100005];
int main()
{
    cin >> n >> m;
    for (int i = 1; i <= n; i++)
        cin >> a[i], sum[i] = sum[i - 1] + a[i];
    for (int i = 1; i <= m; i++)
    {
        cin >> l >> r;
        cout << sum[r] - sum[l - 1] << endl;
    }
    return 0;
}
