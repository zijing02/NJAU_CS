#include <bits/stdc++.h>
using namespace std;
int main()
{
    double x;
    cin >> x;
    double l = 0, r = x;
    while (r - l > 1e-8)
    {
        double mid = (l + r) / 2;
        if (mid * mid >= x)
            r = mid;
        else
            l = mid;
    }
    // tips 题目精度与上面的1e-8有关
    // 6位小数则1e-8，7位小数则1e-9 ......
    printf("%lf/n", l);
    return 0;
}
