#include <bits/stdc++.h>
using namespace std;
// 差分
// 已知数组a1,a2,...,an
// 构造数组b1,b2,...,bn
// 使得ai = b1 + b2 + ... + bi，即ai为b1到bi的前缀和
// 则有bi = ai - a(i-1) (a0 = 0)
// 使得a数组在[l,r]区间每个数加上x
// 则有bl += x, br+1 -= x
// 之后对b数组求前缀和即可得到更新后的a数组

int main()
{
    int n, m, l, r, c, a[100005], b[100005];
    cin >> n >> m;
    for (int i = 1; i <= n; i++)
    {
        cin >> a[i];
        b[i] = a[i] - a[i - 1];
    }
    while (m--)
    {
        cin >> l >> r >> c;
        b[l] += c;
        b[r + 1] -= c;
    }
    for (int i = 1; i <= n; i++)
    {
        b[i] += b[i - 1];
        cout << b[i] << " ";
    }
    return 0;
}
