#include <bits/stdc++.h>
using namespace std;
// 高精度加法
// 使用字符串存储大整数
// 数组开头存储数字个位，末尾存储数字最高位
// 模拟人工加法过程
// 注意：此处实现的是大整数加法，并没有处理负数的情况

vector<int> add(vector<int> &A, vector<int> &B)
{
    vector<int> C;
    int t = 0;

    // 只要A或B没遍历完，或者进位不为0，就继续
    for (int i = 0; i < A.size() || i < B.size() || t; i++)
    {
        // 只要没超过相应数字的长度，就把对应位加上
        // 后续处理进位，t%10是当前位，t/10是更新进位
        if (i < A.size())
            t += A[i];
        if (i < B.size())
            t += B[i];
        C.push_back(t % 10);
        t /= 10;
    }
    return C;
}

int main()
{
    // 读入方法，使用字符串一整行一起读入后反转字符串
    string a, b;
    vector<int> A, B, C;
    cin >> a >> b;
    for (int i = a.size() - 1; i >= 0; i--)
        A.push_back(a[i] - '0');
    for (int i = b.size() - 1; i >= 0; i--)
        B.push_back(b[i] - '0');
    C = add(A, B);
    for (int i = C.size() - 1; i >= 0; i--)
        cout << C[i];
    return 0;
}
