#include <iostream>
using namespace std;
const int N = 1e5 + 10;
int n, k;
int q[N];
int quick_sort(int l, int r, int k)
{
    if (l >= r)
    {
        return q[k - 1];
    }
    int x = q[l];

    int i = l - 1, j = r + 1;
    while (i < j)
    {
        i++;
        while (q[i] < x)
            i++;
        j--;
        while (q[j] > x)
            j--;
        if (i < j)
            swap(q[i], q[j]);
    }

    quick_sort(l, j, k);
    quick_sort(j + 1, r, k);
}

int main()
{
    scanf("%d %d", &n, &k);
    for (int i = 0; i < n; i++)
    {
        scanf("%d", &q[i]);
    }

    cout << quick_sort(0, n - 1, k) << endl;

    return 0;
}
