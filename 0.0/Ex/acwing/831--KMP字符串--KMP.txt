// KMP算法，用于判断一个字符串是否是另一个字符串的子串
// 找出子串中的重复子串，p[1~j] = p[i-j+1~i]
// 最小移动距离和最长重复子串的长度相关
// 是指就是在寻找某一段中  前缀==后缀的最大长度
// 前缀 = 后缀 = 长串中后缀
#include <bits/stdc++.h>
using namespace std;
const int N = 100010, M = 1000010;
int n, m;
char p[N], s[M];
int ne[N];
int main()
{
    cin >> n >> p + 1 >> m >> s + 1;

    // 求ne[j]
    // i从2开始，如果p[1]失败了只能从0开始，所以没有在循环里
    // 从2开始，j=0时，ne[j] = 0
    for (int i = 2, j = 0; i <= n; i++)
    {
        while (j && p[i] != p[j + 1])
            j = ne[j];
        if (p[i] == p[j + 1])
            j++;
        ne[i] = j;
    }

    // 匹配过程
    // 在长串s中参与匹配的是i，在子串p中参与匹配的是j+1
    // 往后移动ne[j]
    for (int i = 1, j = 0; i <= m; i++)
    {
	// 当匹配不上时，子串往后移动，匹配位置往前退
        while (j && s[i] != p[j + 1])
            j = ne[j];
        if (s[i] == p[j + 1])
            j++;
        if (j == n)
        {
            // 匹配成功
            cout << i - n << " ";
            j = ne[j];
        }
    }
    return 0;
}
