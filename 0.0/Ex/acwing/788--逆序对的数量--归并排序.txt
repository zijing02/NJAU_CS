// 逆序对
// 由归并排序延申而来，在进行数组合并时出现相关元素，并进行累加
#include <bits/stdc++.h>
using namespace std;
const int N = 100010;
int a[N], tmp[N];
long long res;
void mergeSort(int *a, int l, int r)
{
    if (l >= r)
        return;
    int mid = (l + r) >> 1;
    mergeSort(a, l, mid);
    mergeSort(a, mid + 1, r);
    int i = l, j = mid + 1, k = 0;
    while (i <= mid && j <= r)
    {
        if (a[i] <= a[j])
            tmp[k++] = a[i++];
        else
        { // 逆序，则从此处到终点全部逆序
            res += mid - i + 1;
            tmp[k++] = a[j++];
        }
    }
    while (i <= mid)
        tmp[k++] = a[i++];
    while (j <= r)
        tmp[k++] = a[j++];
    for (int i = l, j = 0; i <= r; i++, j++)
        a[i] = tmp[j];
}

int main()
{
    int n;
    cin >> n;
    for (int i = 0; i < n; i++)
        cin >> a[i];
    mergeSort(a, 0, n - 1);
    cout << res;
    return 0;
}
