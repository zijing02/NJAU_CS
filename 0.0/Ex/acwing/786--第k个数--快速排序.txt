#include <bits/stdc++.h>
using namespace std;
void quick_sort(int *a, int l, int r)
{
    if (l >= r)
        return;
    int pos_value = a[(l + r) / 2], i = l, j = r;
    while (i < j)
    {
        while (a[i] < pos_value)
            i++;
        while (a[j] > pos_value)
            j--;
        if (i <= j)
            swap(a[i++], a[j--]);
    }
    quick_sort(a, l, j);
    quick_sort(a, i, r);
}
int main()
{
    int a[100005]}, n, k;
    cin >> n >> k;
    for (int i = 0; i < n; i++)
	cin >> a[i];
    quick_sort(a, 0, n - 1);
    // 因为下标从0开始，所以应该输出k-1
    cout << a[k-1];
    return 0;
}

#include <iostream>
using namespace std;
const int N = 1e5 + 10;
int n, k;
int q[N];
int quick_sort(int l, int r, int k)
{
    if (l >= r)
    {
        return q[k - 1];
    }
    int x = q[l];

    int i = l - 1, j = r + 1;
    while (i < j)
    {
        i++;
        while (q[i] < x)
            i++;
        j--;
        while (q[j] > x)
            j--;
        if (i < j)
            swap(q[i], q[j]);
    }

    quick_sort(l, j, k);
    quick_sort(j + 1, r, k);
}

int main()
{
    scanf("%d %d", &n, &k);
    for (int i = 0; i < n; i++)
    {
        scanf("%d", &q[i]);
    }

    cout << quick_sort(0, n - 1, k) << endl;

    return 0;
}
