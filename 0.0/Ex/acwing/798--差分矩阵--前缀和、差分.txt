#include <bits/stdc++.h>
using namespace std;
// 差分矩阵
// 假设有一个矩阵a，定义一个差分矩阵b，a是b的前缀和矩阵
// b[x1][y1] += c
// b[x2+1][y1] -= c
// b[x1][y2+1] -= c
// b[x2+1][y2+1] += c
int n, m, q, a[1005][1005], b[1005][1005];
void insert(int x1, int y1, int x2, int y2, int c)
{
    b[x1][y1] += c;
    b[x2 + 1][y1] -= c;
    b[x1][y2 + 1] -= c;
    b[x2 + 1][y2 + 1] += c;
}
int main()
{
    cin >> n >> m >> q;
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= m; j++)
        {
            cin >> a[i][j];

            // 先把初始矩阵加入差分矩阵
            // 相当于每个点都加了一次
	    // 一开始的b[i][j]就是a[i][j]
            insert(i, j, i, j, a[i][j]);
        }
    }

    while (q--)
    {
        int x1, y1, x2, y2, c;
        cin >> x1 >> y1 >> x2 >> y2 >> c;
        insert(x1, y1, x2, y2, c);
    }

    // 最后求前缀和得a
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= m; j++)
        {
            b[i][j] += b[i - 1][j] + b[i][j - 1] - b[i - 1][j - 1];
            cout << b[i][j] << " ";
        }
        cout << endl;
    }
    return 0;
}
