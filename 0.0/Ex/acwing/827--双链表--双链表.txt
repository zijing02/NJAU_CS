// 双链表，两个指针，一个指向前，一个指向后
#include <bits/stdc++.h>
using namespace std;

const int N = 100010;
int e[N], l[N], r[N], idx;

void init()
{
    // 0表示左端点，1表示右端点
    r[0] = 1;
    l[1] = 0;
    idx = 2;
}

// 在右边插入
void add(int k, int x)
{
    e[idx] = x;
    r[idx] = r[k];
    l[idx] = k;
    l[r[k]] = idx;
    r[k] = idx;
    idx++;
}

// 其实就是在l[k]右边插入
// 两种插入用一种就可以
// void add_left(int k, int x)
// {
//     e[idx] = x;
//     r[idx] = k;
//     l[idx] = l[k];
//     r[l[k]] = idx;
//     l[k] = idx;
//     idx++;
// }

void remove(int k)
{
    r[l[k]] = r[k];
    l[r[k]] = l[k];
}

int main()
{
    int m, k, x;
    string op;
    cin >> m;
    init();
    while (m--)
    {
        cin >> op;
        if (op == "L")
        {
            cin >> x;
            add(0, x);
        }
        else if (op == "R")
        {
            cin >> x;
            add(l[1], x);
        }
        // 下标从2开始所以要加1
        else if (op == "D")
        {
            cin >> k;
            remove(k + 1);
        }
        else if (op == "IL")
        {
            cin >> k >> x;
            // add_left(k+1,x);
            add(l[k + 1], x);
        }
        else if (op == "IR")
        {
            cin >> k >> x;
            add(k + 1, x);
        }
    }
    for (int i = r[0]; i != 1; i = r[i])
        cout << e[i] << " ";
}
