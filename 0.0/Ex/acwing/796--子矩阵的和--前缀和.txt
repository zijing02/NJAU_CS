#include <bits/stdc++.h>
using namespace std;
// 计算前缀和
// s[i][j] = a[1][1] + a[1][2] + ... + a[1][j]
//          + a[2][1] + a[2][2] + ... + a[2][j]
//          ... + a[i][1] + a[i][2] + ... + a[i][j]
// s[i][j] = s[i-1][j] + s[i][j-1] - s[i-1][j-1] + a[i][j]

// 查询子矩阵和(x1,y1)->(x2,y2)
// s[x2][y2] - s[x1-1][y2] - s[x2][y1-1] + s[x1-1][y1-1]
int a[1005][1005], s[1005][1005];
int n, m, q;
int main()
{
    cin >> n >> m >> q;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
        {
            cin >> a[i][j];
            s[i][j] = s[i - 1][j] + s[i][j - 1] - s[i - 1][j - 1] + a[i][j];
        }
    while (q--)
    {
        int x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;
        cout << s[x2][y2] - s[x1 - 1][y2] - s[x2][y1 - 1] + s[x1 - 1][y1 - 1] << endl;
    }
    return 0;
}
