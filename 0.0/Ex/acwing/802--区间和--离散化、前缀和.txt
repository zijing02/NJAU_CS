// 离散化，用于处理数值范围过大，但有效值较少的情况
#include <bits/stdc++.h>
using namespace std;
const int N = 300010;
typedef pair<int, int> PII;
// n表示添加操作数，m表示询问操作数
// a数组表示每个离散化位置的数量
// s数组表示前缀和
int n, m, a[N], s[N];

// insert中存放所有的离散化位置
// add中存放所有的添加操作
// query中存放所有的询问操作
vector<int> alls;
vector<PII> insert, query;
int find(int x)
{
    int l = 0, r = alls.size() - 1;
    while (l < r)
    {
        int mid = l + r >> 1;
        if (alls[mid] >= x)
            r = mid;
        else
            l = mid + 1;
    }

    // 返回离散化后的位置，注意+1,从1开始编号
    return r + 1;
}
int main()
{
    cin >> n >> m;
    for (int i = 0; i < n; i++)
    {
        // x表示位置，c表示数字
        int x, c;
        cin >> x >> c;
        insert.push_back({x, c});
        alls.push_back(x);
    }

    for (int i = 0; i < m; i++)
    {
        int l, r;
        cin >> l >> r;
        query.push_back({l, r});

        // 左右区间加入到离散化数组
        alls.push_back(l);
        alls.push_back(r);
    }

    // 去重
    sort(alls.begin(), alls.end());
    alls.erase(unique(alls.begin(), alls.end()), alls.end());

    for (auto item : insert)
    {
        int x = find(item.first);
        a[x] += item.second;
    }

    // 处理前缀和
    for (int i = 1; i <= alls.size(); i++)
        s[i] = s[i - 1] + a[i];

    // 处理询问
    for (auto item : query)
    {
        int l = find(item.first), r = find(item.second);
        cout << s[r] - s[l - 1] << endl;
    }
    return 0;
}
