// 离散化，用于处理数值范围过大，但有效值较少的情况
#include <bits/stdc++.h>
using namespace std;
const int N = 300010;
typedef pair<int, int> PII;
// n表示添加操作数，m表示询问操作数
// a数组表示每个离散化位置的数量
// s数组表示前缀和
int n, m, a[N], s[N];

// insert中存放所有的离散化位置
// add中存放所有的添加操作
// query中存放所有的询问操作
vector<int> alls;
vector<PII> insert, query;
int find(int x)
{
    int l = 0, r = alls.size() - 1;
    while (l < r)
    {
        int mid = l + r >> 1;
        if (alls[mid] >= x)
            r = mid;
        else
            l = mid + 1;
    }

    // 返回离散化后的位置，注意+1,从1开始编号
    return r + 1;
}
int main()
{
    cin >> n >> m;
    for (int i = 0; i < n; i++)
    {
        // x表示位置，c表示数字
        int x, c;
        cin >> x >> c;
        insert.push_back({x, c});
        alls.push_back(x);
    }

    for (int i = 0; i < m; i++)
    {
        int l, r;
        cin >> l >> r;
        query.push_back({l, r});

        // 左右区间加入到离散化数组
        alls.push_back(l);
        alls.push_back(r);
    }

    // 去重
    sort(alls.begin(), alls.end());
    alls.erase(unique(alls.begin(), alls.end()), alls.end());

    for (auto item : insert)
    {
        int x = find(item.first);
        a[x] += item.second;
    }

    // 处理前缀和
    for (int i = 1; i <= alls.size(); i++)
        s[i] = s[i - 1] + a[i];

    // 处理询问
    for (auto item : query)
    {
        int l = find(item.first), r = find(item.second);
        cout << s[r] - s[l - 1] << endl;
    }
    return 0;
}

// stl偷鸡（繁琐版）
#include <bits/stdc++.h>
using namespace std;
int a[100005], n, m, x, c, l, r;
map<int, pair<int, int>> mp;
bool cmp(const pair<int, pair<int, int>> &a, const pair<int, pair<int, int>> &b)
{
    return a.first < b.first;
}
int main()
{
    cin >> n >> m;
    for (int i = 0; i < n; i++)
    {
        cin >> x >> c;
        mp[x].first = i;
        mp[x].second += c;
    }
    vector < pair<int, pair<int, int>>> vec(mp.begin(), mp.end());
    sort(vec.begin(), vec.end(), cmp);
    for (int i = 0; i < vec.size(); i++)
    {
        vec[i].second.first = i;
    }
    for (int i = 1; i <= vec.size(); i++)
        a[i] = a[i - 1] + vec[i - 1].second.second;
    while (m--)
    {
        cin >> l >> r;
        cout <<  a[upper_bound(vec.begin(), vec.end(), make_pair(r, make_pair(-1, -1)), cmp) - vec.begin()] -a[lower_bound(vec.begin(), vec.end(), make_pair(l, make_pair(-1, -1)), cmp) - vec.begin()] << endl;
    }
    return 0;
}


// stl偷鸡（简化版）
#include <bits/stdc++.h>
using namespace std;
int a[100005], n, m, x, c, l, r;
vector<pair<int, pair<int, int>>> vec;
bool cmp(const pair<int, pair<int, int>> &a, const pair<int, pair<int, int>> &b)
{
    return a.second.first < b.second.first;
}
int main()
{
    cin >> n >> m;
    for (int i = 0; i < n; i++)
    {
        cin >> x >> c;
        vec.push_back({i, {x, c}});
    }
    sort(vec.begin(), vec.end(), cmp);
    for (int i = 0; i < vec.size(); i++)
    {
        vec[i].first = i;
    }
    for (int i = 1; i <= vec.size(); i++)
        a[i] = a[i - 1] + vec[i - 1].second.second;
    while (m--)
    {
        cin >> l >> r;
        cout << a[upper_bound(vec.begin(), vec.end(), make_pair(-1, make_pair(r, -1)), cmp) - vec.begin()] - a[lower_bound(vec.begin(), vec.end(), make_pair(-1, make_pair(l, -1)), cmp) - vec.begin()] << endl;
    }
    return 0;
}
