#include <iostream>
using namespace std;
int n, a[1000005], tmp[1000005];
// 归并排序--分治
// 确定分界点
// 递归左/右
// 合并
void mergeSort(int *a, int l, int r)
{
    if (l >= r)
        return;
    // 逻辑右移相当于除以2
    // int mid = l+r >> 1;
    int mid = (l + r) / 2;
    mergeSort(a, l, mid);
    mergeSort(a, mid + 1, r);

    // 合并,k是已经存放的元素个数，i是左边数组的下标，j是右边数组的下标
    int k = 0, i = l, j = mid + 1;
    while (i <= mid && j <= r)
        if (a[i] <= a[j])
            tmp[k++] = a[i++];
        else
            tmp[k++] = a[j++];

    // 把剩余的元素放进去
    while (i <= mid)
        tmp[k++] = a[i++];
    while (j <= r)
        tmp[k++] = a[j++];

    // 把tmp数组的元素拷贝回a数组
    for (int i = l, j = 0; i <= r; i++, j++)
        a[i] = tmp[j];
}
int main()
{
    cin >> n;
    for (int i = 0; i < n; i++)
    {
        cin >> a[i];
    }
    mergeSort(a, 0, n - 1);
    for (int i = 0; i < n; i++)
    {
        cout << a[i] << " ";
    }

    return 0;
}
