//暴力求解
#include <bits/stdc++.h>
using namespace std;
int removeElement(vector<int> &nums, int val)
{
    int size = nums.size();
    for (int i = 0; i < size; i++)	//此处两个size都是后续自用的size而不是nums的，使用nums的不仅得不到正确结果，还会超时
    {
       if(nums[i] == val){
            for(int j = i; j < size; j++){
                nums[j] = nums[j + 1];
            }
            i--;
            size--;
       }
    }
    return size;
}
int main()
{
    vector<int> a = {0, 1, 2, 2, 3, 0, 4, 2};
    int val = 2;
    cout << removeElement(a, val) << endl;
    for (int i = 0; i < a.size(); i++)
    {
        cout << a[i] << " ";
    }
    return 0;
}



#include <bits/stdc++.h>
using namespace std;
int removeElement(vector<int> &nums, int val)
{
    for (int i = 0; i < nums.size(); i++)
    {
        if (nums[i] == val)
        {
            nums.erase(nums.begin() + i);
            i--;
        }
    }
    return nums.size();
}
int main()
{
    vector<int> a = {0, 1, 2, 2, 3, 0, 4, 2};
    int val = 2;
    cout << removeElement(a, val) << endl;
    for (int i = 0; i < a.size(); i++)
    {
        cout << a[i] << " ";
    }
    return 0;
}

//双指针，快指针遍历数组，慢指针保存数组大小并作更改
#include <bits/stdc++.h>
using namespace std;
int removeElement(vector<int> &nums, int val)
{
    int slowIndex = 0;
    for(int fastIndex = 0; fastIndex < nums.size(); fastIndex++)
    {
        if(nums[fastIndex] != val)
        {
            nums[slowIndex] = nums[fastIndex];
            slowIndex++;
        }
    }
    return slowIndex;
}
int main()
{
    vector<int> a = {0, 1, 2, 2, 3, 0, 4, 2};
    int val = 2;
    cout << removeElement(a, val) << endl;
    for (int i = 0; i < a.size(); i++)
    {
        cout << a[i] << " ";
    }
    return 0;
}
